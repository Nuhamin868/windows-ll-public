cmake_minimum_required(VERSION 3.20)
project(WindowsLL VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
add_library(winll_core 03_SOURCE/core/logger.cpp 03_SOURCE/pe/pe_inspector.cpp 03_SOURCE/hashing/hash_verification.cpp 03_SOURCE/crypto/schema.cpp)
target_include_directories(winll_core PUBLIC 03_SOURCE/include)
target_compile_options(winll_core PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
if(WIN32)
 add_executable(windows_ll 03_SOURCE/core/main.cpp 03_SOURCE/core/application.cpp 03_SOURCE/execution_control/mutex_guard.cpp 03_SOURCE/process/process_inventory.cpp 03_SOURCE/process/process_metadata.cpp 03_SOURCE/retrieval/artifact_client.cpp 03_SOURCE/crypto/artifact_validator.cpp 03_SOURCE/telemetry/event_logger.cpp)
 target_compile_definitions(windows_ll PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
 target_compile_options(windows_ll PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
 target_link_libraries(windows_ll PRIVATE winll_core bcrypt winhttp advapi32)
 target_link_options(windows_ll PRIVATE -municode -static -static-libgcc -static-libstdc++ -Wl,--no-insert-timestamp -Wl,--dynamicbase -Wl,--nxcompat -Wl,--high-entropy-va)
endif()
if(BUILD_TESTING)
 add_executable(winll_unit 04_TESTING/unit/core_tests.cpp)
 target_link_libraries(winll_unit PRIVATE winll_core)
 if(MINGW)
  target_link_options(winll_unit PRIVATE -static -static-libgcc -static-libstdc++ -Wl,--no-insert-timestamp)
 endif()
 add_test(NAME core COMMAND winll_unit "${CMAKE_SOURCE_DIR}/04_TESTING/test_data")
endif()

add_executable(winll_inspect 04_TESTING/unit/inspect_cli.cpp)
target_link_libraries(winll_inspect PRIVATE winll_core)
if(MINGW)
 target_link_options(winll_inspect PRIVATE -static -static-libgcc -static-libstdc++ -Wl,--no-insert-timestamp)
endif()

if(WIN32 AND BUILD_TESTING)
 add_executable(winll_windows_unit 04_TESTING/unit/windows_tests.cpp
  03_SOURCE/execution_control/mutex_guard.cpp 03_SOURCE/crypto/artifact_validator.cpp
  03_SOURCE/retrieval/artifact_client.cpp 03_SOURCE/process/process_metadata.cpp
  03_SOURCE/process/process_inventory.cpp 03_SOURCE/telemetry/event_logger.cpp)
 target_compile_definitions(winll_windows_unit PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00 WINVER=0x0A00)
 target_compile_options(winll_windows_unit PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
 target_link_libraries(winll_windows_unit PRIVATE winll_core bcrypt winhttp advapi32)
 target_link_options(winll_windows_unit PRIVATE -static -static-libgcc -static-libstdc++ -Wl,--no-insert-timestamp)
 add_test(NAME windows_api COMMAND winll_windows_unit "${CMAKE_SOURCE_DIR}/04_TESTING/test_data")
endif()
